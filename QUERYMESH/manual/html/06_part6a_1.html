PART 6A - ADVANCED: agents as FastMCP + A2A servers, tracing, a JWT-protected API, and two user interfaces

This part builds four lines of the QueryMesh project: (6) every agent exposes a FastMCP server with SSE transport and talks to its peers with the A2A protocol, (7) Logfire span tracing and Langfuse Cloud token tracking, (9) a Telegram bot and a Gradio web UI that stream the answer, and the JWT authentication part of line (11).

You start from the result of Part 5: python -m querymesh.graph "..." runs the five agents inside ONE Python process. In this part each agent becomes its own small web server, exactly like microservices in a company.

SECTION 6A.0 - What, why, when

What: five separate servers (retriever 8101, grader 8102, writer 8103, critic 8104, scout 8105). Each one offers its skill twice: as an MCP tool over SSE (the supervisor calls it) and as an A2A agent with an Agent Card (other agents discover it and delegate to it). In front of them sits the API (8090) with JWT login, and two user interfaces call the API.

Why: separate servers can be scaled, restarted and updated one by one (in Part 6B Kubernetes runs each as its own Deployment). Standard protocols (MCP, A2A) mean any MCP- or A2A-speaking program can use your agents, not only your own supervisor.

When: split into services when several people or programs use the system, or when one agent needs more machines than another. For a single script, Part 5's local mode is simpler (and stays available with WORKER_MODE=local).

How the pieces talk:

Gradio / Telegram --HTTP + JWT--> API :8090 --(cache hit? return)--> supervisor (LangGraph)
supervisor --MCP over SSE--> retriever :8101 | grader :8102 | writer :8103 | critic :8104 | scout :8105
retriever --A2A (peer delegation)--> scout          everyone --spans--> Logfire    LLM calls --tokens--> Langfuse

"""A2A helpers: discover agents from their Agent Cards, and delegate work to a peer agent."""
import json

import httpx
from a2a.client import A2ACardResolver, AgentCardResolutionError, ClientConfig, create_client
from a2a.helpers import new_text_message
from a2a.types import Role, SendMessageRequest


async def fetch_card(base_url: str):
    """Download http://agent/.well-known/agent-card.json (the agent's business card)."""
    async with httpx.AsyncClient(timeout=10) as http:
        return await A2ACardResolver(httpx_client=http, base_url=base_url).get_agent_card()


async def discover_agents(base_urls: list[str]) -> dict[str, str]:
    """Return {skill id: MCP SSE URL} for every agent that answers."""
    registry = {}
    for base_url in base_urls:
        card = await fetch_card(base_url)
        for skill in card.skills:
            registry[skill.id] = base_url.rstrip("/") + "/mcp/sse"
    return registry


async def find_peer(base_urls: list[str], skill_id: str) -> str | None:
    """Find the base URL of the agent that offers skill_id (e.g. 'scout')."""
    for base_url in base_urls:
        try:
            card = await fetch_card(base_url)
        except AgentCardResolutionError:  # that agent is down or not an A2A agent: skip it
            continue
        if any(skill.id == skill_id for skill in card.skills):
            return base_url
    return None


async def delegate(base_url: str, payload: dict) -> dict:
    """Send a task to a peer agent over A2A and return its JSON result."""
    card = await fetch_card(base_url)
    client = await create_client(agent=card, client_config=ClientConfig(streaming=False))
    request = SendMessageRequest(message=new_text_message(json.dumps(payload), role=Role.ROLE_USER))
    result: dict = {}
    try:
        async for response in client.send_message(request):
            if response.HasField("task") and response.task.artifacts:
                result = json.loads(response.task.artifacts[0].parts[0].text)
    finally:
        await client.close()
    return result

"""The five agent servers. Each one exposes the same skill twice:
  - as an MCP tool over SSE at /mcp/sse        (the supervisor calls it)
  - as an A2A agent at /a2a + an Agent Card     (other agents discover it and delegate to it)
Run one with:  python -m querymesh.agents retriever
"""
import asyncio
import json
import os
import sys

import uvicorn
from a2a.helpers import get_message_text, new_task_from_user_message, new_text_message, new_text_part
from a2a.server.agent_execution import AgentExecutor, RequestContext
from a2a.server.events import EventQueue
from a2a.server.request_handlers import DefaultRequestHandler
from a2a.server.routes import create_agent_card_routes, create_jsonrpc_routes
from a2a.server.tasks import InMemoryTaskStore, TaskUpdater
from a2a.types import AgentCapabilities, AgentCard, AgentInterface, AgentSkill, TaskState
from fastmcp import FastMCP
from starlette.applications import Starlette
from starlette.responses import JSONResponse
from starlette.routing import Mount, Route

from querymesh import config, observability, skills
from querymesh.discovery import delegate, find_peer

PEER_MIN_HITS = 3  # fewer hits than this -> the retriever asks the scout for help


# ---------- the tools (async wrappers so slow work does not block the server) ----------
async def retrieve(query: str, k: int = 8) -> list[dict]:
    """Hybrid search over indexed arXiv chunks."""
    hits = await asyncio.to_thread(skills.retrieve, query, k)
    if len(hits) < PEER_MIN_HITS:
        scout_url = await find_peer(config.AGENT_URLS, "scout")  # agent discovery
        if scout_url:
            await delegate(scout_url, {"topic": query, "max_papers": 5})  # A2A peer delegation
            hits = await asyncio.to_thread(skills.retrieve, query, k)
    return hits


async def grade(question: str, chunks: list[dict]) -> list[dict]:
    """Keep only the chunks that help answer the question."""
    return await asyncio.to_thread(skills.grade, question, chunks)


async def write(question: str, chunks: list[dict], feedback: str = "") -> dict:
    """Write a cited answer from the chunks."""
    return await asyncio.to_thread(skills.write, question, chunks, feedback)


async def critic(question: str, answer: str, chunks: list[dict]) -> dict:
    """Check that the answer is supported by the chunks."""
    return await asyncio.to_thread(skills.critic, question, answer, chunks)


async def scout(topic: str, max_papers: int = 5) -> dict:
    """Fetch fresh arXiv papers on a topic and index their abstracts."""
    return await asyncio.to_thread(skills.scout, topic, max_papers)


AGENTS = {
    "retriever": {"port": 8101, "tool": retrieve, "about": "Finds arXiv passages with hybrid search"},
    "grader": {"port": 8102, "tool": grade, "about": "Grades passages for relevance"},
    "writer": {"port": 8103, "tool": write, "about": "Writes cited answers"},
    "critic": {"port": 8104, "tool": critic, "about": "Checks answers for hallucinations"},
    "scout": {"port": 8105, "tool": scout, "about": "Fetches fresh papers from arXiv"},
}


# ---------- A2A: how a peer agent's request is executed ----------
class SkillExecutor(AgentExecutor):
    """Reads the A2A message as JSON arguments, runs the tool, returns the JSON result."""

    def __init__(self, tool):
        self.tool = tool

    async def execute(self, context: RequestContext, event_queue: EventQueue) -> None:
        task = context.current_task or new_task_from_user_message(context.message)
        if not context.current_task:
            await event_queue.enqueue_event(task)
        updater = TaskUpdater(event_queue=event_queue, task_id=task.id, context_id=task.context_id)
        arguments = json.loads(get_message_text(context.message) or "{}")
        result = await self.tool(**arguments)
        await updater.add_artifact(parts=[new_text_part(text=json.dumps(result), media_type="application/json")])
        await updater.update_status(state=TaskState.TASK_STATE_COMPLETED, message=new_text_message("done"))

    async def cancel(self, context: RequestContext, event_queue: EventQueue) -> None:
        raise NotImplementedError("cancel is not supported")


# ---------- build one agent server ----------
def build_app(name: str) -> Starlette:
    spec = AGENTS[name]
    skill_id = spec["tool"].__name__
    public_url = os.getenv("PUBLIC_URL", f"http://localhost:{spec['port']}")

    mcp = FastMCP(f"querymesh-{name}")
    mcp.tool(spec["tool"])
    mcp_app = mcp.http_app(transport="sse", path="/sse")

    card = AgentCard(
        name=f"querymesh-{name}",
        description=spec["about"],
        version="1.0.0",
        default_input_modes=["application/json"],
        default_output_modes=["application/json"],
        capabilities=AgentCapabilities(streaming=False),
        supported_interfaces=[AgentInterface(protocol_binding="JSONRPC", url=f"{public_url}/a2a", protocol_version="1.0")],
        skills=[AgentSkill(id=skill_id, name=skill_id, description=spec["about"], tags=["querymesh"])],
    )
    handler = DefaultRequestHandler(agent_executor=SkillExecutor(spec["tool"]),
                                    task_store=InMemoryTaskStore(), agent_card=card)

    async def healthz(request):
        return JSONResponse({"status": "ok", "agent": name})

    routes = [
        Route("/healthz", healthz),
        *create_agent_card_routes(card),
        *create_jsonrpc_routes(handler, "/a2a"),
        Mount("/mcp", app=mcp_app),
    ]
    return Starlette(routes=routes, lifespan=mcp_app.lifespan)


if __name__ == "__main__":
    agent_name = sys.argv[1] if len(sys.argv) > 1 else "retriever"
    observability.setup(f"querymesh-{agent_name}")
    port = int(os.getenv("PORT", AGENTS[agent_name]["port"]))
    uvicorn.run(build_app(agent_name), host="0.0.0.0", port=port)

"""JWT login: POST /token gives a signed token; every protected call must send it."""
from datetime import datetime, timedelta, timezone

import jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer

from querymesh import config

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


def create_token(username: str) -> str:
    now = datetime.now(timezone.utc)
    claims = {"sub": username, "iat": now, "exp": now + timedelta(minutes=config.JWT_MINUTES)}
    return jwt.encode(claims, config.JWT_SECRET, algorithm="HS256")


def check_password(username: str, password: str) -> bool:
    return username == config.API_USERNAME and password == config.API_PASSWORD


def current_user(token: str = Depends(oauth2_scheme)) -> str:
    try:
        claims = jwt.decode(token, config.JWT_SECRET, algorithms=["HS256"])
    except jwt.PyJWTError:
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "invalid or expired token",
                            headers={"WWW-Authenticate": "Bearer"})
    return claims["sub"]

"""The QueryMesh API: login, ask a question, or ask and stream the agents' progress."""
import json

from fastapi import Depends, FastAPI, HTTPException
from fastapi.responses import StreamingResponse
from fastapi.security import OAuth2PasswordRequestForm
from pydantic import BaseModel, Field

from querymesh import cache, observability
from querymesh.auth import check_password, create_token, current_user
from querymesh.graph import stream_answer

app = FastAPI(title="QueryMesh API", version="1.0.0")
observability.setup("querymesh-api", app)


class Question(BaseModel):
    question: str = Field(min_length=3, max_length=500)


@app.get("/healthz")
def healthz() -> dict:
    return {"status": "ok"}


@app.post("/token")
def login(form: OAuth2PasswordRequestForm = Depends()) -> dict:
    if not check_password(form.username, form.password):
        raise HTTPException(401, "wrong username or password")
    return {"access_token": create_token(form.username), "token_type": "bearer"}


@app.post("/ask")
async def ask(body: Question, user: str = Depends(current_user)) -> dict:
    cached = cache.get(body.question)
    if cached:
        return {**cached, "cached": True}
    final: dict = {}
    async for kind, value in stream_answer(body.question):
        if kind == "final":
            final = value
    cache.put(body.question, final)
    return {**final, "cached": False}


@app.post("/ask/stream")
async def ask_stream(body: Question, user: str = Depends(current_user)) -> StreamingResponse:
    """Server-Sent Events: one 'step' event per agent action, then one 'final' event."""

    async def events():
        cached = cache.get(body.question)
        if cached:
            yield f"event: final\ndata: {json.dumps({**cached, 'cached': True})}\n\n"
            return
        async for kind, value in stream_answer(body.question):
            if kind == "final":
                cache.put(body.question, value)
                value = {**value, "cached": False}
            yield f"event: {kind}\ndata: {json.dumps(value)}\n\n"

    return StreamingResponse(events(), media_type="text/event-stream")
