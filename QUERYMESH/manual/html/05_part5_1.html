PART 5 - INTERMEDIATE: Agentic RAG, the supervisor and its five agents, and the answer cache

This part builds three lines of the QueryMesh project: (2) the LangGraph supervisor-worker setup with 5 specialised agents and a shared TypedDict state, (5) agentic RAG: retrieval, then grading, then generation, with retry loops, and (8) the Upstash Redis exact-match cache keyed by SHA-256.

You start from the result of Part 4: OpenSearch on the VM holds arXiv chunks, and python -m querymesh.search "..." returns hybrid results. vLLM from Section 2B.10 is still running on port 8000.

SECTION 5.0 - What, why, when

What: a team of five agents, led by a supervisor, that turns a question into a cited, fact-checked answer. The supervisor follows a loop: retrieve passages, grade them, retry with a better query (or fetch fresh papers) if too few are relevant, write the answer, check it, and rewrite it if the check fails.

Why: plain RAG (search once, then write) fails silently when the search finds the wrong passages or when the model invents a fact. The grading step and the checking step catch both problems, and the loop gives the system a second chance.

When: use agentic RAG when answer quality matters more than speed. Each extra loop costs one or two more LLM calls (a few seconds on our GPU).

The cache: research questions repeat. When two users ask the same question, the second one gets the stored answer in milliseconds, and the GPU does no work.

"""Tracing: Logfire records spans (who called what, how long); Langfuse records LLM tokens."""
import os

import logfire

_configured = False


def setup(service_name: str, app=None) -> None:
    """Call once at the start of every process (API, each agent, UI)."""
    global _configured
    if not _configured:
        logfire.configure(
            service_name=service_name,
            send_to_logfire="if-token-present",  # no LOGFIRE_TOKEN = tracing stays off, no error
            console=False,
        )
        logfire.instrument_httpx()  # every outgoing HTTP call becomes a span
        # MCP tool calls need nothing extra: the MCP SDK (mcp 2) emits OpenTelemetry spans itself
        _configured = True
    if app is not None:
        logfire.instrument_fastapi(app)  # every incoming request becomes a span


def langfuse_callbacks() -> list:
    """LangChain callback that sends prompts, answers and token counts to Langfuse Cloud."""
    if not os.getenv("LANGFUSE_PUBLIC_KEY"):
        return []
    from langfuse.langchain import CallbackHandler

    return [CallbackHandler()]

"""The brains of the five agents. Each skill is a plain Python function you can test alone."""
import re

from pydantic import BaseModel, Field

from querymesh import config, ingest, search
from querymesh.llm import get_llm


def numbered(chunks: list[dict]) -> str:
    return "\n\n".join(
        f"[{n}] ({c['paper_id']}) {c['title']}\n{c['text'][:1200]}"
        for n, c in enumerate(chunks, start=1)
    )


# 1. Retriever agent
def retrieve(query: str, k: int = config.TOP_K) -> list[dict]:
    """Hybrid search (BM25 + vectors + RRF) over the indexed arXiv chunks."""
    return search.hybrid_search(query, k)


# 2. Grader agent
class GradeResult(BaseModel):
    relevant: list[int] = Field(description="Numbers of the passages that help answer the question")


def grade(question: str, chunks: list[dict]) -> list[dict]:
    """Keep only the chunks that really help to answer the question."""
    if not chunks:
        return []
    prompt = (
        "You grade search results for a research assistant.\n"
        f"Question: {question}\n\nPassages:\n{numbered(chunks)}\n\n"
        "Return the numbers of the passages that contain information useful for answering "
        "the question. Return an empty list if none are useful."
    )
    result = get_llm().with_structured_output(GradeResult).invoke(prompt)
    keep = {n for n in result.relevant if 1 <= n <= len(chunks)}
    return [c for n, c in enumerate(chunks, start=1) if n in keep]


# 3. Writer agent
def write(question: str, chunks: list[dict], feedback: str = "") -> dict:
    """Write an answer that cites the passages as [1], [2], ..."""
    prompt = (
        "You are QueryMesh, a research assistant for arXiv papers.\n"
        "Answer the question using ONLY the passages below. After every claim put the "
        "passage number in square brackets, like [2]. If the passages are not enough, say "
        "what is missing. Use at most 250 words.\n\n"
        f"Question: {question}\n\nPassages:\n{numbered(chunks)}\n"
    )
    if feedback:
        prompt += f"\nA reviewer rejected your previous answer for this reason: {feedback}\nFix it.\n"
    answer = get_llm(temperature=0.2).invoke(prompt).content
    cited = sorted({int(n) for n in re.findall(r"\[(\d+)\]", answer) if 1 <= int(n) <= len(chunks)})
    citations = [
        {"n": n, "paper_id": chunks[n - 1]["paper_id"], "title": chunks[n - 1]["title"],
         "url": chunks[n - 1]["url"]}
        for n in cited
    ]
    return {"answer": answer, "citations": citations}


# 4. Critic agent
class Verdict(BaseModel):
    grounded: bool = Field(description="True if every claim is supported by the passages")
    feedback: str = Field(description="If not grounded: which claim is unsupported, else empty")


def critic(question: str, answer: str, chunks: list[dict]) -> dict:
    """Check that the answer only says things the passages support (no hallucination)."""
    prompt = (
        "You are a strict fact checker.\n"
        f"Question: {question}\n\nPassages:\n{numbered(chunks)}\n\nAnswer to check:\n{answer}\n\n"
        "Is every claim in the answer supported by the passages?"
    )
    return get_llm().with_structured_output(Verdict).invoke(prompt).model_dump()


# 5. Scout agent
STOPWORDS = {"what", "which", "when", "where", "does", "about", "with", "from", "that",
             "this", "there", "their", "into", "have", "between", "explain", "compare"}


def topic_to_arxiv_query(topic: str) -> str:
    words = [w for w in re.findall(r"[a-zA-Z0-9-]+", topic.lower())
             if len(w) > 3 and w not in STOPWORDS]
    return " AND ".join(f"all:{w}" for w in words[:4]) or "cat:cs.CL"


def scout(topic: str, max_papers: int = 5) -> dict:
    """Find fresh papers on arXiv right now and index their abstracts."""
    return ingest.ingest_abstracts(topic_to_arxiv_query(topic), max_papers)


# Used by the supervisor itself when the first search finds too little
def rewrite_query(question: str, previous_query: str) -> str:
    prompt = (
        "A search for research papers returned too few relevant passages.\n"
        f"Original question: {question}\nPrevious search query: {previous_query}\n"
        "Write ONE better search query (keywords and synonyms, no explanation)."
    )
    return get_llm(temperature=0.3).invoke(prompt).content.strip().strip('"')[:300]

"""The supervisor-worker graph: one supervisor routes the question through five agents."""
import asyncio
import json
import operator
import sys
from typing import Annotated, TypedDict

import logfire
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command

from querymesh import config, skills


class QueryState(TypedDict, total=False):
    """The shared notebook every agent reads from and writes to."""
    question: str
    search_query: str
    chunks: list[dict]
    relevant: list[dict] | None
    answer: str | None
    citations: list[dict]
    grounded: bool | None
    feedback: str
    retrieval_attempts: int
    generation_attempts: int
    scouted: bool
    steps: Annotated[list[str], operator.add]  # every agent appends a line; nothing is lost


# ---------- how the supervisor reaches an agent ----------
LOCAL_SKILLS = {
    "retrieve": skills.retrieve,
    "grade": skills.grade,
    "write": skills.write,
    "critic": skills.critic,
    "scout": skills.scout,
}
_registry: dict[str, str] = {}  # skill name -> MCP URL, filled by agent discovery


async def call_skill(name: str, **kwargs):
    """local mode: run the Python function. remote mode: call the agent's MCP tool over SSE."""
    with logfire.span("agent call {name}", name=name, mode=config.WORKER_MODE):
        if config.WORKER_MODE == "local":
            return await asyncio.to_thread(LOCAL_SKILLS[name], **kwargs)
        from fastmcp import Client

        from querymesh.discovery import discover_agents

        if not _registry:
            _registry.update(await discover_agents(config.AGENT_URLS))
        async with Client(_registry[name], timeout=300) as mcp:
            result = await mcp.call_tool(name, kwargs)
        return result.data if result.data is not None else json.loads(result.content[0].text)


# ---------- the supervisor ----------
def supervisor(state: QueryState) -> Command:
    """Read the notebook and decide which agent works next (or finish)."""
    if state.get("retrieval_attempts", 0) == 0:
        return Command(goto="retriever", update={"search_query": state["question"]})

    if state.get("answer") is None:
        if state.get("relevant") is None:
            return Command(goto="grader")
        if len(state["relevant"]) >= config.MIN_RELEVANT:
            return Command(goto="writer")
        if state["retrieval_attempts"] < config.MAX_RETRIEVALS:
            better = skills.rewrite_query(state["question"], state["search_query"])
            return Command(goto="retriever", update={
                "search_query": better, "steps": [f"supervisor: search again with '{better}'"]})
        if not state.get("scouted"):
            return Command(goto="scout")
        return Command(goto="writer")  # write with what we have; the writer says what is missing

    if state.get("grounded") is None:
        return Command(goto="critic")
    if not state["grounded"] and state["generation_attempts"] < config.MAX_GENERATIONS:
        return Command(goto="writer")
    return Command(goto=END)


# ---------- the five workers ----------
async def retriever(state: QueryState) -> dict:
    chunks = await call_skill("retrieve", query=state["search_query"], k=config.TOP_K)
    attempts = state.get("retrieval_attempts", 0) + 1
    return {"chunks": chunks, "relevant": None, "retrieval_attempts": attempts,
            "steps": [f"retriever: {len(chunks)} chunks (attempt {attempts})"]}


async def grader(state: QueryState) -> dict:
    relevant = await call_skill("grade", question=state["question"], chunks=state["chunks"])
    return {"relevant": relevant,
            "steps": [f"grader: {len(relevant)} of {len(state['chunks'])} chunks are relevant"]}


async def writer(state: QueryState) -> dict:
    chunks = state.get("relevant") or state.get("chunks", [])
    result = await call_skill("write", question=state["question"], chunks=chunks,
                              feedback=state.get("feedback", ""))
    attempts = state.get("generation_attempts", 0) + 1
    return {"answer": result["answer"], "citations": result["citations"], "grounded": None,
            "generation_attempts": attempts, "steps": [f"writer: answer drafted (attempt {attempts})"]}


async def critic(state: QueryState) -> dict:
    chunks = state.get("relevant") or state.get("chunks", [])
    verdict = await call_skill("critic", question=state["question"], answer=state["answer"],
                               chunks=chunks)
    return {"grounded": verdict["grounded"], "feedback": verdict["feedback"],
            "steps": [f"critic: grounded={verdict['grounded']}"]}


async def scout(state: QueryState) -> dict:
    result = await call_skill("scout", topic=state["question"], max_papers=5)
    return {"scouted": True, "retrieval_attempts": 0, "chunks": [], "relevant": None,
            "steps": [f"scout: added {len(result['added'])} new papers from arXiv"]}


# ---------- wire it together ----------
def build_graph():
    builder = StateGraph(QueryState)
    builder.add_node("supervisor", supervisor, destinations=("retriever", "grader", "writer", "critic", "scout", END))
    for name, node in [("retriever", retriever), ("grader", grader), ("writer", writer),
                       ("critic", critic), ("scout", scout)]:
        builder.add_node(name, node)
        builder.add_edge(name, "supervisor")  # every worker reports back to the supervisor
    builder.add_edge(START, "supervisor")
    return builder.compile()


graph = build_graph()


async def stream_answer(question: str):
    """Yield ('step', text) while the agents work, then ('final', result_dict)."""
    state: dict = {"question": question}
    async for update in graph.astream(state, stream_mode="updates", config={"recursion_limit": 40}):
        for node, change in update.items():
            if not change:
                continue
            state.update({k: v for k, v in change.items() if k != "steps"})
            for line in change.get("steps", []):
                yield "step", line
    yield "final", {"answer": state.get("answer", ""), "citations": state.get("citations", []),
                    "grounded": state.get("grounded")}


async def ask(question: str) -> dict:
    final = {}
    async for kind, value in stream_answer(question):
        if kind == "step":
            print("  ..", value)
        else:
            final = value
    return final


if __name__ == "__main__":
    result = asyncio.run(ask(" ".join(sys.argv[1:]) or "How does reciprocal rank fusion work?"))
    print("\n" + result["answer"] + "\n")
    for c in result["citations"]:

from querymesh import graph, skills

CHUNK = {"paper_id": "2401.00001", "title": "RRF", "text": "RRF adds 1/(60+rank).", "url": "u"}


def fake_skills(monkeypatch, relevant_after_scout=False, grounded_sequence=(True,)):
    calls = []
    grounded = list(grounded_sequence)
    scouted = {"done": False}

    def retrieve(query, k):
        calls.append("retrieve")
        return [CHUNK, CHUNK]

    def grade(question, chunks):
        calls.append("grade")
        return chunks if (not relevant_after_scout or scouted["done"]) else []

    def write(question, chunks, feedback=""):
        calls.append("write")
        return {"answer": "RRF merges rankings [1].", "citations": [{"n": 1, **CHUNK}]}

    def critic(question, answer, chunks):
        calls.append("critic")
        return {"grounded": grounded.pop(0), "feedback": "claim 2 unsupported"}

    def scout(topic, max_papers):
        calls.append("scout")
        scouted["done"] = True
        return {"added": ["2401.00002"]}

    monkeypatch.setattr(graph, "LOCAL_SKILLS", {"retrieve": retrieve, "grade": grade, "write": write,
                                                "critic": critic, "scout": scout})
    monkeypatch.setattr(skills, "rewrite_query", lambda q, p: "better query")
    return calls

    result = await graph.ask("What is RRF?")

"""Exact-match answer cache in Upstash Redis. Key = SHA-256 of the normalised question."""
import hashlib
import json

import logfire
import redis

from querymesh import config

_client = (redis.Redis.from_url(config.REDIS_URL, decode_responses=True, socket_timeout=5)
           if config.REDIS_URL.startswith(("redis://", "rediss://")) else None)  # no URL = no cache


def cache_key(question: str) -> str:
    normalised = " ".join(question.lower().split())  # "What is RAG? " == "what  is rag?"
    digest = hashlib.sha256(f"{config.LLM_MODEL}|{normalised}".encode("utf-8")).hexdigest()
    return f"querymesh:answer:{digest}"


def get(question: str) -> dict | None:
    if _client is None:
        return None
    try:
        raw = _client.get(cache_key(question))
    except redis.RedisError as error:  # the cache must never break the app
        logfire.warn("cache read failed: {error}", error=str(error))
        return None
    return json.loads(raw) if raw else None


def put(question: str, result: dict) -> None:
    if _client is None:
        return
    try:
        _client.set(cache_key(question), json.dumps(result), ex=config.CACHE_TTL_SECONDS)
    except redis.RedisError as error:
        logfire.warn("cache write failed: {error}", error=str(error))

SECTION 5.10 - Common mistake at this level

Mistake: trusting the answer because it sounds confident. A small model writes fluent text even when the passages do not support it.

How to spot it: open the cited paper for one claim and search the page (Ctrl+F) for the key word. If the claim is not there, the Critic missed it. Also look at the steps: "grader: 0 of 8 relevant" followed directly by a writer step (after the scout) means the answer was written from weak evidence.

Fix: keep the Critic step; raise MIN_RELEVANT to 3 in .env for stricter answers; load more papers on the topic (Section 4.5) so the Grader has real evidence to keep.

Second mistake: caching a wrong answer for 24 hours. While you are still tuning prompts, set CACHE_TTL_SECONDS=300 (5 minutes) in .env, or delete keys as in Section 5.9 Step 9.
