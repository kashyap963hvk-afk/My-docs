"""Small client for the QueryMesh API, shared by the Gradio and Telegram interfaces."""
import json

import httpx

from querymesh import config


def get_token() -> str:
    response = httpx.post(f"{config.API_URL}/token", timeout=30,
                          data={"username": config.API_USERNAME, "password": config.API_PASSWORD})
    response.raise_for_status()
    return response.json()["access_token"]


async def stream_question(question: str, token: str):
    """Yield ('step', text) and finally ('final', dict) from the API's Server-Sent Events."""
    headers = {"Authorization": f"Bearer {token}"}
    async with httpx.AsyncClient(timeout=600) as http:
        async with http.stream("POST", f"{config.API_URL}/ask/stream",
                               json={"question": question}, headers=headers) as response:
            response.raise_for_status()
            event = "message"
            async for line in response.aiter_lines():
                if line.startswith("event: "):
                    event = line[len("event: "):]
                elif line.startswith("data: "):
                    yield event, json.loads(line[len("data: "):])


def format_final(final: dict) -> str:
    sources = "\n".join(f"[{c['n']}] {c['title']} - {c['url']}" for c in final.get("citations", []))
    flag = " (from cache)" if final.get("cached") else ""
    return f"{final.get('answer', '')}\n\nSources{flag}:\n{sources or 'none'}"

"""Gradio web chat. Shows each agent step live, then the cited answer."""
import gradio as gr

from querymesh import observability
from querymesh.api_client import format_final, get_token, stream_question

observability.setup("querymesh-gradio")


async def respond(message: str, history: list):
    token = get_token()
    progress = ""
    async for kind, value in stream_question(message, token):
        if kind == "step":
            progress += f"- {value}\n"
            yield f"Working...\n{progress}"
        else:
            yield f"{format_final(value)}\n\nAgent steps:\n{progress}"


demo = gr.ChatInterface(
    fn=respond,
    title="QueryMesh",
    description="Ask a question about recent arXiv papers. Five agents search, grade, write and check.",
    examples=["How does reciprocal rank fusion combine BM25 and vector search?",
              "What are recent methods to reduce hallucination in RAG?"],
)

if __name__ == "__main__":
    demo.launch(server_name="0.0.0.0", server_port=7860)

"""Telegram bot. Send it a question; it edits its reply live while the agents work."""
from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, ContextTypes, MessageHandler, filters

from querymesh import config, observability
from querymesh.api_client import format_final, get_token, stream_question

observability.setup("querymesh-telegram")


async def start(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    await update.message.reply_text("Hi! I am QueryMesh. Send me a question about arXiv research.")


async def answer(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    reply = await update.message.reply_text("Working...")
    progress = ""
    async for kind, value in stream_question(update.message.text, get_token()):
        if kind == "step":
            progress += f"- {value}\n"
            await reply.edit_text(f"Working...\n{progress}")
        else:
            await reply.edit_text(format_final(value)[:4000])  # Telegram limit: 4096 characters


def main() -> None:
    if not config.TELEGRAM_BOT_TOKEN:
        raise SystemExit("TELEGRAM_BOT_TOKEN is empty: put the token from @BotFather in .env")
    app = ApplicationBuilder().token(config.TELEGRAM_BOT_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, answer))
    app.run_polling()


if __name__ == "__main__":
    main()

SECTION 6A.11 - Common mistake at this level

Mistake: changing .env and expecting the running servers to notice. Every process (five agents, API, Gradio, Telegram) reads .env once, at start. The usual symptom: "I added my Logfire token but see no spans" or "the API still uses local mode".

How to spot it: tmux attach -t agents and look at when each window started; compare with when you saved .env.

Fix: restart everything in one go:

tmux kill-session -t agents

Then repeat Section 6A.3 Steps 1-5, Section 6A.7 Step 11, Section 6A.9 Step 11 and Section 6A.10 Step 14. Part 6B replaces these manual windows with docker compose, which restarts all services with one command.

Second mistake: opening ports 8090 or 7860 in the Hyperstack firewall "to make it work". You do not need to: VS Code port forwarding is private and encrypted. Opened ports expose an API and a chat page to the whole internet.
