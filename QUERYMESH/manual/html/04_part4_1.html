PART 4 - BASICS: ingestion pipeline and hybrid search (resume lines 3 and 4)

This part builds the "library": papers come in from arXiv every night, are cut into chunks, turned into vectors and stored in OpenSearch, and one function finds the best chunks for any question. Everything later (the agents, the API, Kubernetes) stands on what you build here.

4.0 What, why, when

What: an ingestion pipeline (arXiv API -> PDF -> Docling -> chunks -> Jina embeddings -> OpenSearch, plus a copy of each PDF in S3) and a hybrid search function (BM25 + vectors merged with RRF), scheduled every night by an Airflow DAG.

Why: an LLM alone does not know last week's papers and makes up citations. With a search index of real papers, every answer can point to a real paper.

When: you run the full pipeline once by hand in this part, then Airflow runs it every night at 02:00 UTC. The hybrid search function is what the Retriever agent calls in Part 5.

"""Turn text into vectors with the Jina AI embeddings API."""
import httpx

from querymesh import config

JINA_URL = "https://api.jina.ai/v1/embeddings"
BATCH_SIZE = 32


def embed(texts: list[str], task: str) -> list[list[float]]:
    """task = "retrieval.passage" for paper chunks, "retrieval.query" for questions."""
    if not config.JINA_API_KEY:
        raise RuntimeError("JINA_API_KEY is empty: put your key in the .env file")
    vectors: list[list[float]] = []
    for start in range(0, len(texts), BATCH_SIZE):
        batch = texts[start:start + BATCH_SIZE]
        response = httpx.post(
            JINA_URL,
            headers={"Authorization": f"Bearer {config.JINA_API_KEY}"},
            json={
                "model": config.EMBED_MODEL,
                "task": task,
                "dimensions": config.EMBED_DIM,
                "normalized": True,
                "input": batch,
            },
            timeout=120,
        )
        response.raise_for_status()
        rows = sorted(response.json()["data"], key=lambda row: row["index"])
        vectors.extend(row["embedding"] for row in rows)
    return vectors


if __name__ == "__main__":
    vector = embed(["hybrid search combines keywords and meaning"], "retrieval.query")[0]
    print(f"dimensions={len(vector)} first_values={[round(v, 4) for v in vector[:5]]}")

"""OpenSearch: create the index, store chunks, and run hybrid (BM25 + vector + RRF) search."""
import sys

from opensearchpy import OpenSearch, helpers

from querymesh import config
from querymesh.embeddings import embed

client = OpenSearch(hosts=[config.OPENSEARCH_URL], timeout=60)

INDEX_BODY = {
    "settings": {"index": {"knn": True, "number_of_shards": 1, "number_of_replicas": 0}},
    "mappings": {
        "properties": {
            "chunk_id": {"type": "keyword"},
            "paper_id": {"type": "keyword"},
            "title": {"type": "text"},
            "authors": {"type": "keyword"},
            "published": {"type": "date"},
            "url": {"type": "keyword"},
            "text": {"type": "text", "analyzer": "english"},
            "embedding": {
                "type": "knn_vector",
                "dimension": config.EMBED_DIM,
                "method": {"name": "hnsw", "engine": "lucene", "space_type": "cosinesimil"},
            },
        }
    },
}

PIPELINE_BODY = {
    "description": "QueryMesh hybrid search: merge BM25 and vector results with Reciprocal Rank Fusion",
    "phase_results_processors": [
        {"score-ranker-processor": {"combination": {"technique": "rrf", "rank_constant": 60}}}
    ],
}


def setup() -> None:
    """Create the index and the RRF search pipeline (safe to run many times)."""
    if not client.indices.exists(index=config.INDEX_NAME):
        client.indices.create(index=config.INDEX_NAME, body=INDEX_BODY)
        print(f"created index {config.INDEX_NAME}")
    client.transport.perform_request(
        "PUT", f"/_search/pipeline/{config.SEARCH_PIPELINE}", body=PIPELINE_BODY
    )
    print(f"search pipeline {config.SEARCH_PIPELINE} is ready")


def paper_exists(paper_id: str) -> bool:
    result = client.count(index=config.INDEX_NAME, body={"query": {"term": {"paper_id": paper_id}}})
    return result["count"] > 0


def index_chunks(chunks: list[dict]) -> int:
    """Embed every chunk and store it. Each chunk needs chunk_id, paper_id, title, text..."""
    if not chunks:
        return 0
    vectors = embed([c["title"] + "\n" + c["text"] for c in chunks], "retrieval.passage")
    actions = [
        {"_index": config.INDEX_NAME, "_id": c["chunk_id"], "_source": {**c, "embedding": v}}
        for c, v in zip(chunks, vectors)
    ]
    ok, _ = helpers.bulk(client, actions, refresh=True)
    return ok


def hybrid_search(query: str, k: int = config.TOP_K) -> list[dict]:
    """Keyword search + meaning search, merged by RRF. Returns the best k chunks."""
    vector = embed([query], "retrieval.query")[0]
    body = {
        "size": k,
        "_source": {"excludes": ["embedding"]},
        "query": {
            "hybrid": {
                "queries": [
                    {"multi_match": {"query": query, "fields": ["title^2", "text"]}},
                    {"knn": {"embedding": {"vector": vector, "k": k * 5}}},
                ]
            }
        },
    }
    response = client.search(
        index=config.INDEX_NAME, body=body, params={"search_pipeline": config.SEARCH_PIPELINE}
    )
    return [{**hit["_source"], "score": hit["_score"]} for hit in response["hits"]["hits"]]


if __name__ == "__main__":
    if sys.argv[1:] == ["setup"]:
        setup()
    else:
        question = " ".join(sys.argv[1:]) or "What is retrieval-augmented generation?"
        for n, hit in enumerate(hybrid_search(question, k=5), start=1):
            print(f"[{n}] {hit['score']:.4f} {hit['paper_id']} {hit['title'][:70]}")
            print("    " + hit["text"][:160].replace("\n", " "))

"""Ingestion: arXiv API -> PDF -> Docling -> chunks -> Jina vectors -> OpenSearch (+ S3 copy)."""
import argparse
import re
from pathlib import Path

import arxiv
import boto3
import httpx

from querymesh import config, search

DATA_DIR = Path("data/pdfs")
CHUNK_CHARS = 1500
OVERLAP_CHARS = 200
_converter = None  # Docling is slow to load, so we create it once, the first time we need it


def fetch_papers(query: str, max_papers: int) -> list[dict]:
    """Ask the arXiv API for the newest papers that match the query."""
    arxiv_client = arxiv.Client(page_size=min(max_papers, 100), delay_seconds=3.0, num_retries=3)
    search_spec = arxiv.Search(
        query=query, max_results=max_papers, sort_by=arxiv.SortCriterion.SubmittedDate
    )
    papers = []
    for result in arxiv_client.results(search_spec):
        papers.append({
            "paper_id": result.get_short_id(),
            "title": " ".join(result.title.split()),
            "authors": [a.name for a in result.authors][:10],
            "published": result.published.date().isoformat(),
            "url": result.entry_id,
            "pdf_url": result.pdf_url,
            "summary": " ".join(result.summary.split()),
        })
    return papers


def download_pdf(paper: dict) -> Path:
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    path = DATA_DIR / f"{paper['paper_id'].replace('/', '_')}.pdf"
    if not path.exists():
        response = httpx.get(paper["pdf_url"], follow_redirects=True, timeout=120)
        response.raise_for_status()
        path.write_bytes(response.content)
    return path


def pdf_to_markdown(path: Path) -> str:
    """Docling reads the PDF layout (headings, paragraphs, tables) and returns Markdown."""
    global _converter
    if _converter is None:
        from docling.datamodel.base_models import InputFormat
        from docling.datamodel.pipeline_options import PdfPipelineOptions
        from docling.document_converter import DocumentConverter, PdfFormatOption

        options = PdfPipelineOptions(do_ocr=False, do_table_structure=True)
        _converter = DocumentConverter(
            format_options={InputFormat.PDF: PdfFormatOption(pipeline_options=options)}
        )
    return _converter.convert(str(path)).document.export_to_markdown()


def chunk_markdown(markdown: str) -> list[str]:
    """Split by headings, then cut long sections into overlapping windows."""
    markdown = markdown.split("\n## References")[0]  # the reference list is noise for search
    sections = [s.strip() for s in re.split(r"\n(?=#{1,3} )", markdown) if s.strip()]
    chunks = []
    for section in sections:
        start = 0
        while start < len(section):
            piece = section[start:start + CHUNK_CHARS].strip()
            if len(piece) > 100:
                chunks.append(piece)
            start += CHUNK_CHARS - OVERLAP_CHARS
    return chunks


def upload_pdf(path: Path) -> None:
    """Keep the original PDF in S3 (FakeCloud or AWS). AWS_ENDPOINT_URL decides which one."""
    if config.PAPERS_BUCKET:
        boto3.client("s3").upload_file(str(path), config.PAPERS_BUCKET, f"pdfs/{path.name}")


def make_chunks(paper: dict, texts: list[str]) -> list[dict]:
    keep = ("paper_id", "title", "authors", "published", "url")
    return [
        {**{k: paper[k] for k in keep}, "chunk_id": f"{paper['paper_id']}-{n}", "text": text}
        for n, text in enumerate(texts)
    ]


def ingest_full_text(query: str, max_papers: int) -> dict:
    """Nightly job: full PDFs, parsed by Docling."""
    search.setup()
    added, skipped = [], []
    for paper in fetch_papers(query, max_papers):
        if search.paper_exists(paper["paper_id"]):
            skipped.append(paper["paper_id"])
            continue
        path = download_pdf(paper)
        texts = chunk_markdown(pdf_to_markdown(path))
        count = search.index_chunks(make_chunks(paper, texts))
        upload_pdf(path)
        added.append(paper["paper_id"])
        print(f"indexed {paper['paper_id']}: {count} chunks  {paper['title'][:60]}")
    return {"added": added, "skipped": skipped}


def ingest_abstracts(query: str, max_papers: int) -> dict:
    """Fast path used by the Scout agent: index title + abstract only (no PDF)."""
    search.setup()
    added = []
    for paper in fetch_papers(query, max_papers):
        if search.paper_exists(paper["paper_id"]):
            continue
        search.index_chunks(make_chunks(paper, [paper["summary"]]))
        added.append(paper["paper_id"])
    return {"added": added}


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Load arXiv papers into OpenSearch")
    parser.add_argument("--query", default="cat:cs.CL AND abs:retrieval")
    parser.add_argument("--max-papers", type=int, default=5)
    parser.add_argument("--abstracts-only", action="store_true")
    args = parser.parse_args()
    run = ingest_abstracts if args.abstracts_only else ingest_full_text
    print(run(args.query, args.max_papers))

"""Airflow DAG: every day at 02:00 UTC, load the newest arXiv papers into QueryMesh."""
from datetime import datetime

from airflow.providers.standard.operators.bash import BashOperator
from airflow.sdk import dag

PROJECT = "/home/ubuntu/querymesh"
TOPICS = {
    "rag": "cat:cs.CL AND abs:retrieval",
    "agents": "cat:cs.AI AND abs:agent",
}


@dag(
    dag_id="querymesh_ingest",
    schedule="0 2 * * *",
    start_date=datetime(2026, 10, 1),
    catchup=False,
    tags=["querymesh"],
)
def querymesh_ingest():
    previous = None
    for name, query in TOPICS.items():
        task = BashOperator(
            task_id=f"ingest_{name}",
            bash_command=(
                f"cd {PROJECT} && .venv/bin/python -m querymesh.ingest "
                f"--query '{query}' --max-papers 10"
            ),
            retries=2,
        )
        if previous:
            previous >> task  # one topic after the other: arXiv allows 1 request per 3 seconds
        previous = task


querymesh_ingest()

4.9 Common mistake at this level

Mixing up the two embedding tasks, or changing the embedding model after data is indexed. Symptom: search returns chunks that have nothing to do with the question, although everything "works" and there are no errors. How to spot it: run the BM25-only query from Section 4.7 Step 4; if BM25 finds the right paper but hybrid_search does not rank it in the top 5, the vectors are wrong. Fix: make sure chunks use retrieval.passage and questions use retrieval.query (they do in our code), and if you ever change EMBED_MODEL or EMBED_DIM, delete the index (Section 4.2 Watch out) and ingest again.

A second classic mistake: running ingestion with the aws CLI pointed at FakeCloud but .env pointed at real AWS (or the reverse). Symptom: aws s3 ls shows no PDFs although ingestion succeeded. Fix: compare echo $AWS_ENDPOINT_URL with the AWS_ENDPOINT_URL line in .env (Section 4.4).
